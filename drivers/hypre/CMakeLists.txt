cmake_minimum_required(VERSION 3.24)
project(sma_hypre_driver CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(HYPRE REQUIRED CONFIG)
# Parameter tables generated from the installed headers this driver is compiled against.
get_target_property(_hypre_inc HYPRE::HYPRE INTERFACE_INCLUDE_DIRECTORIES)
find_file(HYPRE_UTILITIES_H HYPRE_utilities.h PATHS ${_hypre_inc} NO_DEFAULT_PATH REQUIRED)
find_file(HYPRE_KRYLOV_H HYPRE_krylov.h PATHS ${_hypre_inc} NO_DEFAULT_PATH REQUIRED)
find_file(HYPRE_PARCSR_LS_H HYPRE_parcsr_ls.h PATHS ${_hypre_inc} NO_DEFAULT_PATH REQUIRED)
set(_gen ${CMAKE_CURRENT_BINARY_DIR}/gen)
add_custom_command(OUTPUT ${_gen}/hypre_params.inc ${_gen}/hypre_params.json
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/gen_params.py ${HYPRE_UTILITIES_H} ${HYPRE_KRYLOV_H} ${HYPRE_PARCSR_LS_H} ${_gen}
  DEPENDS gen_params.py ${HYPRE_UTILITIES_H} ${HYPRE_KRYLOV_H} ${HYPRE_PARCSR_LS_H})
add_executable(sma_hypre hypre_driver.cpp ${_gen}/hypre_params.inc)
target_include_directories(sma_hypre PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common ${_gen})
target_link_libraries(sma_hypre PRIVATE HYPRE::HYPRE MPI::MPI_CXX CUDA::cudart $<$<BOOL:${OpenMP_CXX_FOUND}>:OpenMP::OpenMP_CXX>)
install(TARGETS sma_hypre)
install(FILES ${_gen}/hypre_params.json DESTINATION share/sma/params)
