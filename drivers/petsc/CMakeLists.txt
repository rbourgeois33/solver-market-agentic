cmake_minimum_required(VERSION 3.24)
project(sma_petsc_driver CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PETSC REQUIRED IMPORTED_TARGET PETSc)
# libcuda: needed by Kokkos; on GPU-less build nodes CMake resolves it to the CUDA stub library.
add_executable(sma_petsc petsc_driver.cpp)
target_include_directories(sma_petsc PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../common)
target_link_libraries(sma_petsc PRIVATE PkgConfig::PETSC MPI::MPI_CXX CUDA::cudart CUDA::cuda_driver $<$<BOOL:${OpenMP_CXX_FOUND}>:OpenMP::OpenMP_CXX>)
install(TARGETS sma_petsc)
